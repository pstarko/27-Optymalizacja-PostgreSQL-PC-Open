-- 1. Najpierw włączamy rozszerzenie pgvector
CREATE EXTENSION IF NOT EXISTS vector;

-- 2. Tworzymy tabelę z przykładowymi wektorami
CREATE TABLE IF NOT EXISTS przykladowe_wektory (
    id SERIAL PRIMARY KEY,
    nazwa TEXT,
    wektor VECTOR(3)  -- 3-wymiarowe wektory dla łatwości zrozumienia
);


-- 3. Wstawiamy przykładowe dane (punkty w przestrzeni 3D)
INSERT INTO przykladowe_wektory (nazwa, wektor) VALUES
    ('Punkt A', '[1, 2, 3]'),
    ('Punkt B', '[4, 5, 6]'),
    ('Punkt C', '[1, 2, 4]'),  -- bardzo podobny do A
    ('Punkt D', '[10, 20, 30]'), -- daleko od wszystkich
    ('Punkt E', '[1.1, 2.1, 3.1]'); -- bardzo blisko A
    
 -- 4. Sprawdźmy nasze dane   
select * from przykladowe_wektory


-- 5. KLUCZOWY PRZYKŁAD: Obliczamy odległości euklidesowe od Punktu A
SELECT
  p.nazwa,
  p.wektor,
  p.wektor <-> '[1, 2, 3]'::vector AS odleglosc_od_A
FROM przykladowe_wektory p
ORDER BY p.wektor <-> '[1, 2, 3]'::vector;

-- 6. Znajdźmy 3 najbliższe punkty do Punktu A
explain
SELECT 
    nazwa,
    wektor,
    wektor <-> '[1, 2, 3]'::vector AS odleglosc
FROM przykladowe_wektory
WHERE nazwa != 'Punkt A'  -- wykluczamy sam punkt A
ORDER BY wektor <-> '[1, 2, 3]'::vector
LIMIT 3;

-- 7. Porównanie różnych metryk
SELECT 
    nazwa,
    wektor,
    -- Odległość euklidesowa (L2)
    wektor <-> '[1, 2, 3]'::vector AS euklidesowa,
    -- Podobieństwo kosinusowe (dla porównania)
    1 - (wektor <=> '[1, 2, 3]'::vector) AS podobienstwo_cosinusowe
FROM przykladowe_wektory
ORDER BY wektor <-> '[1, 2, 3]'::vector;


--8. PRAKTYCZNY PRZYKŁAD: Symulacja wyszukiwania podobnych dokumentów
-- Załóżmy, że mamy osadzenia tekstów (embeddings)
CREATE TABLE IF NOT EXISTS dokumenty (
    id SERIAL PRIMARY KEY,
    tytul TEXT,
    embedding VECTOR(5)  -- uproszczone 5-wymiarowe osadzenie
);


INSERT INTO dokumenty (tytul, embedding) VALUES
    ('Artykuł o kotach', '[0.8, 0.2, 0.1, 0.9, 0.3]'),
    ('Tekst o psach', '[0.7, 0.3, 0.2, 0.8, 0.4]'),
    ('Artykuł o samochodach', '[0.1, 0.9, 0.8, 0.2, 0.7]'),
    ('Tekst o kotach domowych', '[0.85, 0.15, 0.05, 0.95, 0.25]'),
    ('Artykuł o motorach', '[0.2, 0.8, 0.9, 0.1, 0.8]');

SELECT 
    tytul,
    embedding <-> '[0.8, 0.2, 0.1, 0.9, 0.3]'::vector AS odleglosc_euklidesowa
FROM dokumenty
ORDER BY embedding <-> '[0.8, 0.2, 0.1, 0.9, 0.3]'::vector
LIMIT 3;


--Cosinus
CREATE TABLE produkty (
    id SERIAL PRIMARY KEY,
    nazwa TEXT,
    opis TEXT,
    cechy_vector VECTOR(3)  -- 3-wymiarowy wektor dla prostoty
);

-- Każdy wektor reprezentuje cechy produktu: [cena, jakość, popularność]
INSERT INTO produkty (nazwa, opis, cechy_vector) VALUES
    ('Laptop Gaming', 'Wysokiej klasy laptop do gier', '[0.9, 0.8, 0.7]'),
    ('Laptop Biurowy', 'Podstawowy laptop do pracy', '[0.3, 0.6, 0.8]'),
    ('Laptop Premium', 'Drogi laptop biznesowy', '[0.95, 0.9, 0.6]'),
    ('Tablet Podstawowy', 'Tani tablet do internetu', '[0.2, 0.4, 0.9]'),
    ('Tablet Pro', 'Profesjonalny tablet graficzny', '[0.8, 0.85, 0.5]');


-- Szukamy produktów podobnych do wektora zapytania [0.85, 0.8, 0.6]
-- (reprezentuje: wysoka cena, wysoka jakość, średnia popularność)

SELECT 
    nazwa,
    opis,
    cechy_vector,
    -- Obliczamy podobieństwo kosinusowe (1 - cosine distance)
    1 - (cechy_vector <=> '[0.85, 0.8, 0.6]') as podobienstwo_kosinusowe,
    -- Dla porównania - odległość euklidesowa
    cechy_vector <-> '[0.85, 0.8, 0.6]' as odleglosc_euklidesowa
FROM produkty
ORDER BY cechy_vector <=> '[0.85, 0.8, 0.6]'  -- Sortujemy po odległości kosinusowej
LIMIT 5;
