DROP TABLE IF EXISTS planner_stats_demo;

CREATE TABLE planner_stats_demo (
    order_id    BIGSERIAL PRIMARY KEY,
    status      TEXT NOT NULL,
    order_date  DATE NOT NULL,
    total_amount NUMERIC(10,2) NOT NULL
);

SELECT setseed(0.42);


INSERT INTO planner_stats_demo (
    status,
    order_date,
    total_amount
)
    SELECT
        CASE
            WHEN random() < 0.95 THEN 'completed'
            ELSE 'cancelled'
        END,
        CURRENT_DATE - floor(random() * 365)::int,
        round((25 + random() * 975)::numeric, 2)
    FROM generate_series(1, 300000);
