--Deoptymalizacja przez użycie funkcji zamiast bezpośredniego kodu

--tworzymy funkcję
CREATE OR REPLACE FUNCTION par_fun(val int)
RETURNS numeric AS $$
DECLARE
    result numeric;
BEGIN
    SELECT SUM(unitPrice) INTO result
    FROM SalesOrderDetail
    WHERE SalesOrderDetailID = val;
    
    RETURN result;
END;
$$ LANGUAGE plpgsql;


--COST: 9385, Time: 27.468
EXPLAIN (ANALYZE, BUFFERS)
select 
	SalesOrderDetailID,
	sum(unitPrice) over (partition by salesorderdetailid) as suma
	from SalesOrderDetail
	limit 1000
	

--COST 274, Time: 1087
select 
	SalesOrderDetailID,
	par_fun(SalesOrderDetailID)
	from SalesOrderDetail
limit 1000

/*
Różnica między kosztem (cost) i rzeczywistym czasem wykonania (execution time) zapytań 
w PostgreSQL wynika z tego, jak planner estymuje koszt, a jak faktycznie jest wykonane zapytanie.

Wyjaśnienie różnicy w twoim przykładzie:
Koszt (cost) to szacunkowa wartość, oparta na statystykach tabel i modelu kosztów, 
która odzwierciedla liczbę jednostek pracy (np. liczba odczytów stron dysku). 
Jest to wartość arbitralna, służy plannerowi do wyboru planu, ale nie 
jest dokładnym czasem wykonania.

Execution time to rzeczywisty czas wykonania zapytania zmierzony po jego uruchomieniu.

***Dlaczego zapytanie z funkcją ma niższy koszt, ale dłuższy czas wykonania?
Funkcja par_fun(SalesOrderDetailID) może być czarną skrzynką dla planera — 
czyli planner nie może dobrze oszacować jej kosztu, więc przypisuje jej niski koszt. 
Jednak w rzeczywistości funkcja ta może być kosztowna obliczeniowo 
(np. skomplikowana logika, petle, wolne operacje), stąd długi czas wykonania.

W pierwszym zapytaniu jest funkcja agregująca okna sum(unitPrice) OVER (...), 
która ma precyzyjnie wyliczony koszt na podstawie znanych operatorów w plannerze, 
choć jej koszt jest większy, ale jest efektywnie wykonywana, 
np. korzystając z wewnętrznych optymalizacji PostgreSQL, więc wykonuje się szybciej.

Koszt nie uwzględnia wszystkich aspektów, np. kosztu CPU, kosztów funkcji 
zdefiniowanych przez użytkownika czy wpływów pamięci podręcznej.

***Co warto zrobić?
	Sprawdź szczegóły funkcji par_fun pod kątem optymalności (czy nie robi nadmiarowych pętli, 
	zapytań, obliczeń).

	Zaktualizuj statystyki tabel poleceniem ANALYZE SalesOrderDetail.

	Pamiętaj, aby przy porównywaniu planów nie kierować się tylko liczbowym kosztem — 
	dużo ważniejszy jest rzeczywisty czas wykonania oraz profilowanie zapytania.

	Podsumowując, niższy koszt w planie to nie zawsze szybsze wykonanie, 
	zwłaszcza jeśli zapytanie lub funkcja jest złożona i planner nie potrafi dobrze 
	oszacować rzeczywistego nakładu pracy
*/

--szersze sprawdzenie funkcji
CREATE EXTENSION pg_stat_statements;

/*
Kolumna						Co oznacza
query						Tekst samego zapytania SQL
calls						Ile razy to zapytanie zostało wykonane
blk_read_time				Całkowity czas (w ms) spędzony na czytaniu bloków danych z dysku
blk_write_time				Całkowity czas (w ms) spędzony na pisaniu bloków danych na dysk
shared_blks_read			Liczba bloków przeczytanych fizycznie z dysku (najdroższe operacje)
shared_blks_hit				Liczba bloków znalezionych w shared buffer cache (szybkie, z RAM) 
 
 
Cache Hit Ratio (ważne):
Możesz obliczyć efektywność:

cache_hit_ratio = shared_blks_hit / (shared_blks_hit + shared_blks_read)
              = 85000 / (85000 + 15000) 
              = 85% 
>90% = dobra efektywność cache
<50% = funkcja zbyt dużo czyta z dysku, możesz optymalizować indeksy
 */
 
 
SELECT *
FROM pg_stat_statements
WHERE query LIKE '%par_fun%';
