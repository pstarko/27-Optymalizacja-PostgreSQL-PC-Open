SELECT attname, format_type(atttypid, atttypmod) AS typ
FROM pg_attribute
WHERE attrelid = 'AllData'::regclass
  AND atttypid IN (
    SELECT oid FROM pg_type WHERE typname IN ('text', 'bytea', 'jsonb', 'varchar', 'aclitem', 'xml')
  )
  AND attnum > 0
  AND NOT attisdropped;
